# Parliament, cabinet and presidency lifecycle never reset polling or seats.
TFR_EST_suspend_parliament = {
    TFR_EST_hide_executive_display = yes
    clr_country_flag = TFR_EST_parliament_open
    remove_ideas = TFR_EST_constitutional_presidency
    remove_ideas = TFR_EST_presidency_karis
    remove_ideas = TFR_EST_coalition_narrow
    remove_ideas = TFR_EST_coalition_working
    remove_ideas = TFR_EST_coalition_broad
    remove_ideas = TFR_EST_caretaker_cabinet
}

TFR_EST_reconcile_presidency = {
    if = {
        limit = { TFR_EST_independent_parliament = yes }
        if = {
            limit = { has_country_flag = TFR_EST_karis_president }
            remove_ideas = TFR_EST_constitutional_presidency
            if = { limit = { NOT = { has_idea = TFR_EST_presidency_karis } } add_ideas = TFR_EST_presidency_karis }
        }
        else = {
            remove_ideas = TFR_EST_presidency_karis
            if = { limit = { NOT = { has_idea = TFR_EST_constitutional_presidency } } add_ideas = TFR_EST_constitutional_presidency }
        }
    }
    TFR_EST_reconcile_executive_display = yes
}

# Native elections are disabled because the party ledger schedules elections.
# HOI4 requires ruling_party even when only changing elections_allowed.
# Keep the confirmed PM's ideology, including during caretaker government.
TFR_EST_sync_native_cabinet_politics = {
    if = {
        limit = { TFR_EST_independent_parliament = yes }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 1 compare = equals } }
            set_politics = { ruling_party = market_liberal elections_allowed = no name = TFR_EST_party_reform long_name = TFR_EST_party_reform }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 2 compare = equals } }
            set_politics = { ruling_party = social_liberal elections_allowed = no name = TFR_EST_party_centre long_name = TFR_EST_party_centre }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 3 compare = equals } }
            set_politics = { ruling_party = authoritarian_democrat elections_allowed = no name = TFR_EST_party_ekre long_name = TFR_EST_party_ekre }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 4 compare = equals } }
            set_politics = { ruling_party = conservative elections_allowed = no name = TFR_EST_party_isamaa long_name = TFR_EST_party_isamaa }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 5 compare = equals } }
            set_politics = { ruling_party = social_democrat elections_allowed = no name = TFR_EST_party_sde long_name = TFR_EST_party_sde }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 6 compare = equals } }
            set_politics = { ruling_party = social_liberal elections_allowed = no name = TFR_EST_party_e200 long_name = TFR_EST_party_e200 }
        }
        if = {
            limit = { check_variable = { var = TFR_EST_pm_party value = 7 compare = equals } }
            set_politics = { ruling_party = market_liberal elections_allowed = no name = TFR_EST_party_pp long_name = TFR_EST_party_pp }
        }
    }
}

# The confirmed party ledger owns the head_minister slot and ruling ideology.
# The native country leader displays the separate constitutional president.
TFR_EST_reconcile_cabinet_leadership = {
    if = {
        limit = { TFR_EST_independent_parliament = yes }

        if = { limit = { check_variable = { var = TFR_EST_pm_party value = 5 compare = equals } } set_party_name = { ideology = social_democrat name = TFR_EST_party_sde long_name = TFR_EST_party_sde } }

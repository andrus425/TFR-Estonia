# Install once and update immediately after changing a backing variable.
# Neither occupation nor restoration erases the country's accumulated values.
TFR_EST_refresh_country_modifiers = {
    if = {
        limit = { tag = EST }
        if = {
            limit = { NOT = { has_dynamic_modifier = { modifier = TFR_EST_economy_dynamic } } }
            add_dynamic_modifier = { modifier = TFR_EST_economy_dynamic }
        }
        if = {
            limit = { NOT = { has_dynamic_modifier = { modifier = TFR_EST_military_dynamic } } }
            add_dynamic_modifier = { modifier = TFR_EST_military_dynamic }
        }
        force_update_dynamic_modifier = yes
    }
}

# Runs only after committing a valid 101-seat election and its plurality.
# A later valid vote ends this programme even if Isamaa wins again.
TFR_EST_note_isamaa_2023_election = {
    if = {
        limit = { has_country_flag = TFR_EST_isamaa_2023_mandate }
        set_country_flag = TFR_EST_isamaa_2023_mandate_ended
    }
    if = {
        limit = {
            date > 2023.3.4
            date < 2024.1.1
            check_variable = { var = TFR_EST_mandate_party value = 4 compare = equals }
            NOT = { has_country_flag = TFR_EST_postwar_election_pending }
            NOT = { has_country_flag = TFR_EST_postwar_election_held }
            NOT = { has_country_flag = TFR_EST_isamaa_won_2023_election }
        }
        set_country_flag = TFR_EST_isamaa_won_2023_election
        set_country_flag = TFR_EST_isamaa_2023_mandate
        clr_country_flag = TFR_EST_isamaa_2023_mandate_ended
    }
}
